; Keyboard & mouse as a controller for Minecraft Dungeons II.
; Edit, then restart the game. The on-screen key list is built from this file.
;
; A key is a letter or digit (W), a name (Space, Enter, Escape, Tab, Shift,
; Ctrl, Alt, Backspace, CapsLock, Up, Down, Left, Right, Backtick, F1-F12,
; Mouse1-Mouse5), None, or a virtual-key code (0x20). Several keys per line are allowed,
; comma-separated (up to three). Mouse1 = left, Mouse2 = right, Mouse3 = middle, Mouse4/5 = side.
;
; The keys are the game's own keyboard defaults (Settings > Controls > Keyboard),
; each pressing the controller button the game uses for that action. Only the
; menu wheel moves, from S (a movement key here) to Tab.

[Mouse]
; Moving the mouse this far (pixels) after letting go of the movement keys
; switches to mouse mode (normal cursor and keyboard). 0 = off.
MouseModeOnMovePx=40
; 0: moving the mouse never leaves controller mode; only the cursor key (Alt),
; the menu keys or the toggle key (`) give you the mouse.
MouseMoveSwitches=1
; 1: when moving the mouse brings the cursor back, it starts from the middle of
; the window. 0: it appears where the mouse moved it.
CursorFromMiddle=0
; A small mouse bump (under MouseModeOnMovePx) flips the game to keyboard mode;
; the mod answers with a light right-stick nudge (percent) for a moment to flip it
; back. Lower it if it ever makes you dodge; 0 = off.
BumpNudgePct=30
; Keep the mouse inside the game window in controller mode (so a click can't
; land in another app): the hidden cursor is moved back to the middle when it
; drifts far. Backtick, Alt or switching apps lets it out.
KeepCursorInWindow=1
; Holding a mouse button bound to RT (the bow) switches to keyboard mode so the
; game aims the bow at the cursor; WASD doesn't switch back until you let go.
BowAimsWithMouse=1

[Keys]
; The game's own menu shortcuts (its defaults: I inventory, M map, J quests,
; F social, G emotes, U collectibles, K event log). They go straight to the game and switch
; to mouse mode (normal cursor, clicking stays in mouse mode). Any fight key
; (movement, jump, abilities) switches back to controller mode. The menus can
; also be put on the controller's buttons instead ([Buttons] DUp...; the key
; layout editor switches each menu between keyboard and controller).
MenuKeys=I, M, J, F, G, U, K, Escape
; In mouse mode these close the menu (the game gets them) and switch straight
; back to controller mode.
BackKeys=Escape
; Controller buttons that open a menu: once released, switch to mouse mode.
; Pressing the same key again (e.g. E) closes the menu and stays in controller mode.
MouseAfter=None
; Only a tap (shorter than this) leaves you in mouse mode. Holding longer opens a
; quick overlay instead (hold I = mini inventory, hold G = emote wheel):
; you're back in controller mode as soon as you let go.
MenuTapMs=300
; Typing mode: starts typing (this key itself isn't typed); every key then
; reaches the game, Enter included, until Esc (which the game doesn't get).
TypeKey=T
; Keys that always reach the game unchanged, without switching to mouse mode:
; the game's teleport statue (Z), teleport to player 1-4 (F1-F4) and teleport to
; player (X).
PassKeys=Z, F1, F2, F3, F4, X
; The game's own keys for actions you moved to other keys: they never reach the
; game, in any mode (typing aside). The key layout editor fills this in.
DisabledKeys=Q, Shift
; Game keys of instant actions (teleports): a key remapped onto one doesn't switch
; to mouse mode. The key layout editor fills this in.
InstantKeys=
; Hold for the normal mouse cursor (like Genshin's Alt). On a Mac this is the
; Option key: install.command tells CrossOver to send Option as Alt.
Cursor=Alt
; Hold = the cursor shows (in the middle) while the key is held; letting go hides
; it and goes back to the controller, like Genshin. Toggle = press once to show
; it, press again to hide it. A game menu that's open keeps the cursor.
CursorMode=Hold
; Switches the mod off/on until pressed again.
Toggle=Backtick
; Shows or hides the key list. (F1-F4 are the game's teleport keys.)
Legend=F9

[Remap]
